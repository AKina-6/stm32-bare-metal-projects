cmake_minimum_required(VERSION 3.16)
project(layered_firmware C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/W4)
else()
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(app_core src/app/app.c)
target_include_directories(app_core PUBLIC include)
add_library(host_platform src/platform/host/host_platform.c)
target_include_directories(host_platform PUBLIC include src/platform/host)
add_executable(layered_firmware_sim src/platform/host/host_main.c)
target_link_libraries(layered_firmware_sim PRIVATE app_core host_platform)
enable_testing()
add_executable(test_app tests/test_app.c)
target_link_libraries(test_app PRIVATE app_core host_platform)
add_test(NAME app_layer_tests COMMAND test_app)
