cmake_minimum_required(VERSION 3.16)
project(servo_control C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
 add_compile_options(/W4)
else()
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(servo_core src/servo_controller.c)
target_include_directories(servo_core PUBLIC include)
add_executable(servo_control_sim src/host_main.c)
target_link_libraries(servo_control_sim PRIVATE servo_core)
enable_testing()
add_executable(test_servo tests/test_servo.c)
target_link_libraries(test_servo PRIVATE servo_core)
add_test(NAME servo_controller_tests COMMAND test_servo)
