cmake_minimum_required(VERSION 3.16)
project(flash_settings C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(settings_core src/settings/settings.c src/settings/crc32.c src/settings/settings_record.c src/app/settings_manager.c)
target_include_directories(settings_core PUBLIC include)
add_library(host_flash src/platform/host/host_flash.c)
target_include_directories(host_flash PUBLIC include src/platform/host)
add_executable(flash_settings_sim src/platform/host/host_main.c)
target_link_libraries(flash_settings_sim PRIVATE settings_core host_flash)
enable_testing()
add_executable(test_settings_record tests/test_settings_record.c)
target_link_libraries(test_settings_record PRIVATE settings_core)
add_test(NAME settings_record_tests COMMAND test_settings_record)
add_executable(test_settings_manager tests/test_settings_manager.c)
target_link_libraries(test_settings_manager PRIVATE settings_core host_flash)
add_test(NAME settings_manager_tests COMMAND test_settings_manager)
