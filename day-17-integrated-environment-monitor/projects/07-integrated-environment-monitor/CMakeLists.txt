cmake_minimum_required(VERSION 3.16)
project(integrated_environment_monitor C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
 add_compile_options(/W4)
else()
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(environment_core src/app/environment_app.c src/app/telemetry_formatter.c src/app/ui_model.c)
target_include_directories(environment_core PUBLIC include)
add_executable(environment_monitor_sim src/platform/host/host_main.c)
target_link_libraries(environment_monitor_sim PRIVATE environment_core)
enable_testing()
add_executable(test_environment_app tests/test_environment_app.c)
target_link_libraries(test_environment_app PRIVATE environment_core)
add_test(NAME environment_app_tests COMMAND test_environment_app)
add_executable(test_formatters tests/test_formatters.c)
target_link_libraries(test_formatters PRIVATE environment_core)
add_test(NAME formatter_tests COMMAND test_formatters)
